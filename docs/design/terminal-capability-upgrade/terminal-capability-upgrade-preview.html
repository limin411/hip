配置示例

hip.toml 终端配置
# 终端配置
[terminal]
shell = "default"
colorTheme = "follow"
bell = "visual"

# 渲染设置
[terminal.rendering]
webgl = true # GPU 加速
ligatures = true # 字体连字
unicodeVersion = "11" # Unicode 版本

# 协议设置
[terminal.protocol]
hyperlinks = true # OSC 8 超链接
clipboardRead = "ask" # 剪贴板读取策略

# 快捷键自定义
[terminal.keybindings]
copy = "Ctrl+Shift+C"
paste = "Ctrl+Shift+V"
search = "Ctrl+F"
newTab = "Ctrl+T"
closeTab = "Ctrl+W"
nextTab = "Ctrl+Tab"
splitVertical = "Ctrl+\\"
splitHorizontal = "Ctrl+Shift+\\"
